cmake_minimum_required(VERSION 3.24)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(SpeedRacerGSBridge C CXX ASM)
    enable_testing()
endif()
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "The experimental GS bridge requires Windows x64")
endif()
if(NOT TARGET kzgs)
    set(SRGS_KZGS_SOURCE_DIR "" CACHE PATH "Pinned, compatible kzgs source/build adapter")
    if(NOT EXISTS "${SRGS_KZGS_SOURCE_DIR}/CMakeLists.txt")
        message(FATAL_ERROR "Configure SRGS_KZGS_SOURCE_DIR before building the GS bridge")
    endif()
    set(KZGS_BUILD_TEST OFF CACHE BOOL "" FORCE)
    set(KZGS_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
    add_subdirectory("${SRGS_KZGS_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/kzgs")
endif()
add_library(SpeedRacerGS SHARED bridge.cpp)
set_target_properties(SpeedRacerGS PROPERTIES PREFIX "")
target_compile_features(SpeedRacerGS PRIVATE cxx_std_20)
target_link_libraries(SpeedRacerGS PRIVATE kzgs kzgs_flags)
target_include_directories(SpeedRacerGS PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
kzgs_stage_resources(SpeedRacerGS)
if(MINGW)
    get_filename_component(_gs_toolchain_bin "${CMAKE_CXX_COMPILER}" DIRECTORY)
    foreach(_gs_runtime libgcc_s_seh-1.dll libwinpthread-1.dll libstdc++-6.dll)
        if(NOT EXISTS "${_gs_toolchain_bin}/${_gs_runtime}")
            message(FATAL_ERROR "Missing native GS compiler runtime: ${_gs_runtime}")
        endif()
        add_custom_command(TARGET SpeedRacerGS POST_BUILD
            COMMAND ${CMAKE_COMMAND} -E copy_if_different "${_gs_toolchain_bin}/${_gs_runtime}" "$<TARGET_FILE_DIR:SpeedRacerGS>"
            VERBATIM)
    endforeach()
endif()

add_executable(gs_hw_bridge_test "${CMAKE_CURRENT_SOURCE_DIR}/../../tests/gs_hw_bridge.c")
set_target_properties(gs_hw_bridge_test PROPERTIES C_STANDARD 11)
target_include_directories(gs_hw_bridge_test PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
add_test(NAME gs_hw_bridge_c COMMAND gs_hw_bridge_test "$<TARGET_FILE:SpeedRacerGS>")
set_tests_properties(gs_hw_bridge_c PROPERTIES TIMEOUT 90)
add_test(NAME gs_hw_bridge_headless COMMAND gs_hw_bridge_test "$<TARGET_FILE:SpeedRacerGS>" --headless)
set_tests_properties(gs_hw_bridge_headless PROPERTIES TIMEOUT 90)
add_executable(gs_hw_packets_test "${CMAKE_CURRENT_SOURCE_DIR}/../../tests/gs_hw_packets.cpp")
target_compile_features(gs_hw_packets_test PRIVATE cxx_std_20)
target_include_directories(gs_hw_packets_test PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
add_test(NAME gs_hw_packets COMMAND gs_hw_packets_test)

add_executable(gs_hw_filter_test "${CMAKE_CURRENT_SOURCE_DIR}/../../tests/gs_hw_filter.c")
set_target_properties(gs_hw_filter_test PROPERTIES C_STANDARD 11)
target_include_directories(gs_hw_filter_test PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
add_test(NAME gs_hw_filter COMMAND gs_hw_filter_test "$<TARGET_FILE:SpeedRacerGS>")
set_tests_properties(gs_hw_filter PROPERTIES TIMEOUT 90)
